package featurecat.lizzie.analysis;
import java.nio.file.Path;
import org.json.*;
public class Parity {
 public static void main(String[] args) throws Exception {
  var ctor=XGBoostStrengthModel.Features.class.getDeclaredConstructor(double[].class);ctor.setAccessible(true);
  JSONArray result=new JSONArray();
  for(int i=0;i<40;i++) {double[] f=new double[29];for(int j=0;j<29;j++)f[j]=((i*37+j*19)%101)/100.0;
   var features=ctor.newInstance((Object)f);
   double p=XGBoostStrengthModel.predictXGBoost20TunRankValue(features,Path.of(args[0]),Path.of(args[1])).getAsDouble();
   result.put(new JSONObject().put("features",f).put("prediction",p));
  }
  System.out.println(result);
 }
}
